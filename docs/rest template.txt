```java
public List<ProductDto> findProductsByCategoryId(Long categoryId) {
    // 1. On récupère les produits de notre base locale
    List<Product> products = productRepository.findByCategoryId(categoryId);
    
    // 2. On récupère le nom de la catégorie UNE SEULE FOIS pour toute la liste
    // (Utilisation de ta méthode fetchCategoryName déjà créée)
    String categoryName = fetchCategoryName(categoryId);

    // 3. Transformation en DTOs
    List<ProductDto> productDtos = new ArrayList<>();
    products.forEach(product -> {
        ProductDto dto = mapEntityToDto(product);
        // On force le nom récupéré pour éviter des appels réseaux redondants dans le mapper
        dto.setCategoryName(categoryName); 
        productDtos.add(dto);
    });

    return productDtos;
}

```
